in this lecture we learn to handle form

const App = () => {

  const submitHandler=(e)=>{
    e.preventDefault()
    console.log('form submitted')
  }
  return (
    <div>
      <form onSubmit={(e)=>{
        submitHandler(e)
      }}>
      <input type="text" placeholder='Enter your name' />
      <button>Submit</button>
      </form>
    </div>
  )
}
